/** * OPERATION ELF SCANNER APP - backend API * ------------------------------------------------------- * Backend only. The actual scanner page (with camera) lives on a separate * website (e.g. GitHub Pages) because Apps Script's own pages run inside a * sandboxed iframe that browsers block camera access from. * * Access control: real Google Sign-In. The external page sends a Google * ID token with every request; this script verifies it against Google's * servers, then checks the signed-in email against CONFIG.TEAM below. */ const CONFIG = { SHEET_NAME: 'Sheet1', CODE_COLUMN: 'Code', BARCODE_COLUMN: 'Code', // matched against what's actually printed on barcodes now LOCATION_COLUMN: 'Location', // "Current Location" in the UI REC_SITE_COLUMN: 'Rec Site', // "Dropoff Location" in the UI PACKING_OPTIONS: [ 'Packing Day (Check-in)', 'Packing Day (Spirit)', 'Packing Day (Packing)', 'Packing Day (Yellow Table)', 'Packing Day (Blue Table)', 'Packing Day (Pink Table)', 'Packing Day (Green Table)', 'Packing Day (Truck)' ], STATUS_COLUMN: 'Status', STATUS_OPTIONS: ['Empty', 'Pending', 'Received'], NOTES_COLUMN: 'Sitenotes', COMPLETE_COLUMN: 'Complete', COMPLETE_OPTIONS: ['Empty', 'Partial', 'Complete'], MISSING_COLUMN: 'Missing', DETAIL_FIELDS: ['Age', 'Gender', 'Wish', 'Color', 'Shirt', 'Pants', 'Shoes', 'GC', 'Notes'], LOG_UPDATED_AT: true, UPDATED_AT_COLUMN: 'Last Updated At', UPDATED_BY_COLUMN: 'Last Updated By', // ---- Google Sign-In ---- GOOGLE_CLIENT_ID: '533971898412-nhovuse7b5db3bq2n7t5b01552shj8ke.apps.googleusercontent.com', // Roster lives in a separate tab (see TEAM_SHEET_NAME) instead of hardcoded // here, so you can add/remove people day-of without touching this file. TEAM_SHEET_NAME: 'Team', TEAM_EMAIL_COLUMN: 'Email', TEAM_ROLE_COLUMN: 'Role', // Which views each role is allowed to use. Enforced server-side on every // request that touches a specific view, not just hidden in the UI. ROLE_VIEWS: { admin: ['sitehost', 'packing', 'guiding', 'workshop'], sitehost: ['sitehost', 'packing', 'guiding'], volunteer: ['packing', 'guiding'], }, // Per-role field whitelist for 'lookup' responses, so the browser never // receives sensitive columns (Email, Phone, etc.) a given role doesn't need. LOOKUP_FIELDS_BY_VIEW: { sitehost: ['Code', 'Status', 'Sitenotes', 'Complete', 'Missing', 'Age', 'Gender', 'Wish', 'Color', 'Shirt', 'Pants', 'Shoes', 'GC', 'Notes'], guiding: ['Code', 'Missing', 'Age', 'Gender', 'Wish', 'Color', 'Shirt', 'Pants', 'Shoes', 'GC', 'Notes', 'Packing Table'], packing: ['Complete', 'Packing Table'], }, }; function hasViewAccess_(role, view) { const allowed = CONFIG.ROLE_VIEWS[role] || []; return allowed.indexOf(view) !== -1; } function doGet(e) { const callback = (e.parameter.callback || 'callback').replace(/[^a-zA-Z0-9_]/g, ''); let payload; try { const auth = verifyAuth_(e.parameter.idToken); if (!auth.valid) { payload = { success: false, error: auth.error, authError: true }; } else { switch (e.parameter.action) { case 'config': { const sheet = getSheet_(); const headers = getHeaders_(sheet); payload = { success: true, role: auth.role, email: auth.email, recSiteOptions: getUniqueColumnValues_(sheet, headers, CONFIG.REC_SITE_COLUMN), packingOptions: CONFIG.PACKING_OPTIONS, statusOptions: CONFIG.STATUS_OPTIONS, completeOptions: CONFIG.COMPLETE_OPTIONS, detailFields: CONFIG.DETAIL_FIELDS, }; break; } case 'lookup': if (!hasViewAccess_(auth.role, e.parameter.view)) { payload = { success: false, error: 'Not authorized for this view.' }; } else { payload = lookupItem_(e.parameter.barcode, e.parameter.view); } break; case 'update': if (!hasViewAccess_(auth.role, e.parameter.view)) { payload = { success: false, error: 'Not authorized for this view.' }; } else { payload = updateItem_(e.parameter.barcode, e.parameter.field, e.parameter.value, auth.email); } break; case 'updateFields': if (!hasViewAccess_(auth.role, 'sitehost')) { payload = { success: false, error: 'Not authorized for Site Host Elf.' }; } else { payload = updateFields_(e.parameter.barcode, JSON.parse(e.parameter.fields || '{}'), auth.email); } break; case 'summary': if (!hasViewAccess_(auth.role, 'workshop')) { payload = { success: false, error: "Not authorized for Santa's Workshop." }; } else { payload = getSummary_(e.parameter.location || 'All', e.parameter.site || 'All', e.parameter.dropoff || 'All'); } break; default: payload = { success: false, error: 'Unknown action.' }; } } } catch (err) { payload = { success: false, error: err.message }; } const body = callback + '(' + JSON.stringify(payload) + ')'; return ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JAVASCRIPT); } /** * Verifies a Google ID token by asking Google's own tokeninfo endpoint to * validate its signature, then checks the signed-in email against CONFIG.TEAM. * * Performance: the same ID token is reused for every request during a * person's session, but re-verifying against Google's servers on every * single scan adds several seconds of network latency each time. We cache * a successful verification for 5 minutes (well under the token's own ~1hr * lifetime), so only the first request in a while pays that network cost. */ function verifyAuth_(idToken) { if (!idToken) return { valid: false, error: 'Not signed in.' }; const cache = CacheService.getScriptCache(); const cacheKey = 'auth_' + Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, idToken) .map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join(''); const cached = cache.get(cacheKey); if (cached) return JSON.parse(cached); let info; try { const resp = UrlFetchApp.fetch( 'https://oauth2.googleapis.com/tokeninfo?id_token=' + encodeURIComponent(idToken), { muteHttpExceptions: true } ); if (resp.getResponseCode() !== 200) { return { valid: false, error: 'Sign-in expired. Please sign in again.' }; } info = JSON.parse(resp.getContentText()); } catch (err) { return { valid: false, error: 'Could not verify sign-in.' }; } if (info.aud !== CONFIG.GOOGLE_CLIENT_ID) return { valid: false, error: 'Token was not issued for this app.' }; if (info.email_verified !== 'true' && info.email_verified !== true) return { valid: false, error: 'Email not verified by Google.' }; const email = String(info.email || '').toLowerCase().trim(); const role = getTeamRole_(email); if (!role) return { valid: false, error: 'This Google account is not approved. Contact your coordinator.' }; const result = { valid: true, email: email, role: role }; cache.put(cacheKey, JSON.stringify(result), 300); // 5 minutes return result; } /** * Looks up a person's role from the "Team" sheet tab (Email | Role columns), * so the roster can be edited live without touching this file or redeploying. * The whole roster is cached for 60 seconds to avoid re-reading the sheet on * every sign-in check; new additions take effect within about a minute. */ function getTeamRole_(email) { const cache = CacheService.getScriptCache(); const cached = cache.get('team_roster'); let roster; if (cached) { roster = JSON.parse(cached); } else { roster = {}; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(CONFIG.TEAM_SHEET_NAME); if (sheet) { const headers = getHeaders_(sheet); const emailCol = headers.indexOf(CONFIG.TEAM_EMAIL_COLUMN); const roleCol = headers.indexOf(CONFIG.TEAM_ROLE_COLUMN); const lastRow = sheet.getLastRow(); if (emailCol !== -1 && roleCol !== -1 && lastRow > 1) { const rows = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues(); rows.forEach(row => { const e = String(row[emailCol] || '').toLowerCase().trim(); const r = String(row[roleCol] || '').toLowerCase().trim(); if (e && r) roster[e] = r; }); } } cache.put('team_roster', JSON.stringify(roster), 60); // 1 minute } return roster[email] || null; } function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(CONFIG.SHEET_NAME); if (!sheet) throw new Error('Sheet "' + CONFIG.SHEET_NAME + '" not found.'); return sheet; } function getHeaders_(sheet) { return sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; } function extractSitePrefix_(codeVal) { const m = String(codeVal || '').match(/^[A-Za-z]+/); return m ? m[0].toUpperCase() : ''; } /** Sorts "Host Site 2" before "Host Site 10" instead of plain string order. */ function naturalCompare_(a, b) { return String(a).localeCompare(String(b), undefined, { numeric: true, sensitivity: 'base' }); } function getUniqueColumnValues_(sheet, headers, columnName) { const col = headers.indexOf(columnName); if (col === -1) return []; const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; const values = sheet.getRange(2, col + 1, lastRow - 1, 1).getValues(); const set = {}; values.forEach(r => { const v = String(r[0] || '').trim(); if (v) set[v] = true; }); return Object.keys(set).sort(naturalCompare_); } /** * view (optional): when given, only the whitelisted fields for that role are * returned to the browser (see CONFIG.LOOKUP_FIELDS_BY_VIEW). Internal calls * (from updateItem_/updateFields_) omit view and get the full row, since that * data never leaves the server. * * Performance: searches only the barcode-match column first (one thin read), then * fetches just the single matching row, instead of reading the entire sheet * on every lookup. Much faster and more consistent as the sheet grows. */ function lookupItem_(barcode, view) { const sheet = getSheet_(); const headers = getHeaders_(sheet); const barcodeCol = headers.indexOf(CONFIG.BARCODE_COLUMN); if (barcodeCol === -1) throw new Error('Barcode column "' + CONFIG.BARCODE_COLUMN + '" not found.'); const lastRow = sheet.getLastRow(); if (lastRow < 2) return { success: true, found: false }; const barcodeColumn = sheet.getRange(2, barcodeCol + 1, lastRow - 1, 1).getValues(); const target = String(barcode).trim().toUpperCase(); let matchIndex = -1; for (let i = 0; i < barcodeColumn.length; i++) { if (String(barcodeColumn[i][0]).trim().toUpperCase() === target) { matchIndex = i; break; } } if (matchIndex === -1) return { success: true, found: false }; const rowNumber = matchIndex + 2; const rowValues = sheet.getRange(rowNumber, 1, 1, sheet.getLastColumn()).getValues()[0]; const allowedFields = view ? CONFIG.LOOKUP_FIELDS_BY_VIEW[view] : null; const row = {}; headers.forEach((h, idx) => { if (!allowedFields || allowedFields.indexOf(h) !== -1) row[h] = rowValues[idx]; }); return { success: true, found: true, rowNumber: rowNumber, data: row }; } function stampLogColumns_(sheet, headers, rowNumber, email) { if (!CONFIG.LOG_UPDATED_AT) return; const userCol = headers.indexOf(CONFIG.UPDATED_BY_COLUMN); const timeCol = headers.indexOf(CONFIG.UPDATED_AT_COLUMN); if (userCol !== -1) sheet.getRange(rowNumber, userCol + 1).setValue(email || 'Unknown'); if (timeCol !== -1) sheet.getRange(rowNumber, timeCol + 1).setValue(new Date()); } function updateItem_(barcode, field, value, email) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (e) { return { success: false, error: 'System busy, please try again in a moment.' }; } try { const sheet = getSheet_(); const headers = getHeaders_(sheet); const lookup = lookupItem_(barcode); if (!lookup.found) return { success: false, error: 'Item not found for barcode: ' + barcode }; const fieldCol = headers.indexOf(field); if (fieldCol === -1) return { success: false, error: 'Field "' + field + '" not found in sheet.' }; sheet.getRange(lookup.rowNumber, fieldCol + 1).setValue(value); stampLogColumns_(sheet, headers, lookup.rowNumber, email); return { success: true, message: 'Updated ' + field + ' to ' + value }; } finally { lock.releaseLock(); } } function updateFields_(barcode, fields, email) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (e) { return { success: false, error: 'System busy, please try again in a moment.' }; } try { const sheet = getSheet_(); const headers = getHeaders_(sheet); const lookup = lookupItem_(barcode); if (!lookup.found) return { success: false, error: 'Item not found for barcode: ' + barcode }; const skipped = []; Object.keys(fields).forEach(fieldName => { const col = headers.indexOf(fieldName); if (col === -1) { skipped.push(fieldName); return; } sheet.getRange(lookup.rowNumber, col + 1).setValue(fields[fieldName]); }); stampLogColumns_(sheet, headers, lookup.rowNumber, email); return { success: true, message: 'Updated ' + (Object.keys(fields).length - skipped.length) + ' field(s)', skipped: skipped }; } finally { lock.releaseLock(); } } function getSummary_(currentLocationFilter, siteFilter, dropoffFilter) { const sheet = getSheet_(); const headers = getHeaders_(sheet); const codeCol = headers.indexOf(CONFIG.CODE_COLUMN); const locCol = headers.indexOf(CONFIG.LOCATION_COLUMN); const recCol = headers.indexOf(CONFIG.REC_SITE_COLUMN); if (locCol === -1) throw new Error('Location column not found.'); if (codeCol === -1) throw new Error('Code column not found.'); if (recCol === -1) throw new Error('Rec Site column not found.'); const lastRow = sheet.getLastRow(); const data = lastRow > 1 ? sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues() : []; const siteCounts = {}; const packingCounts = {}; let siteLocationTotal = 0; let totalWishes = 0; const codes = []; const sitePrefixesSet = {}; const recSiteSet = {}; data.forEach(row => { const codeVal = row[codeCol]; const prefix = extractSitePrefix_(codeVal); if (prefix) sitePrefixesSet[prefix] = true; const recVal = String(row[recCol] || '').trim(); if (recVal) recSiteSet[recVal] = true; }); data.forEach(row => { const codeVal = row[codeCol]; const prefix = extractSitePrefix_(codeVal); if (siteFilter !== 'All' && prefix !== siteFilter) return; const recVal = String(row[recCol] || '').trim(); if (dropoffFilter !== 'All' && recVal !== dropoffFilter) return; const locVal = String(row[locCol] || '').trim(); if (currentLocationFilter === 'Not Received') { if (locVal !== '') return; } else if (currentLocationFilter !== 'All' && locVal !== currentLocationFilter) { return; } totalWishes++; codes.push(codeVal); const isPackingLoc = CONFIG.PACKING_OPTIONS.indexOf(locVal) !== -1; if (recVal) { if (!siteCounts[recVal]) siteCounts[recVal] = { rec: 0, loc: 0 }; siteCounts[recVal].rec++; } if (locVal && isPackingLoc) { packingCounts[locVal] = (packingCounts[locVal] || 0) + 1; } else if (locVal) { if (!siteCounts[locVal]) siteCounts[locVal] = { rec: 0, loc: 0 }; siteCounts[locVal].loc++; siteLocationTotal++; } }); const packingTotal = Object.keys(packingCounts).reduce((s, k) => s + packingCounts[k], 0); const allSitesLocationTotal = siteLocationTotal + packingTotal; return { success: true, siteCounts: siteCounts, packingCounts: packingCounts, allSitesLocationTotal: allSitesLocationTotal, totalWishes: totalWishes, codes: codes, sitePrefixes: Object.keys(sitePrefixesSet).sort(naturalCompare_), recSiteOptions: Object.keys(recSiteSet).sort(naturalCompare_) };
